cmake_minimum_required(VERSION 3.20)
project(SekiModAssistantNative VERSION 0.3.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Windows 10/11 x64 target is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(SekiModAssistant WIN32 main.cpp app.rc)
add_executable(SekiModSetup WIN32 installer.cpp setup.rc)
set_target_properties(SekiModSetup PROPERTIES OUTPUT_NAME SekiModSetupStub)
target_compile_definitions(SekiModAssistant PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00 NOMINMAX)
target_link_libraries(SekiModAssistant PRIVATE bcrypt shell32 ole32 uuid advapi32 comctl32 wininet user32 gdi32 gdiplus dwmapi)
target_compile_definitions(SekiModSetup PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00 NOMINMAX)
target_link_libraries(SekiModSetup PRIVATE bcrypt shell32 ole32 uuid advapi32 comctl32 wininet user32 gdi32 gdiplus dwmapi uxtheme)
if(MSVC)
  target_compile_options(SekiModAssistant PRIVATE /W4 /utf-8 /EHsc)
  target_compile_options(SekiModSetup PRIVATE /W4 /utf-8 /EHsc)
else()
  target_compile_options(SekiModAssistant PRIVATE -Wall -Wextra)
  target_compile_options(SekiModSetup PRIVATE -Wall -Wextra)
  target_link_options(SekiModAssistant PRIVATE -municode -static -static-libgcc -static-libstdc++)
  target_link_options(SekiModSetup PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
install(TARGETS SekiModAssistant RUNTIME DESTINATION .)
